- 2 patterns of user-based CF
- re-visit

The candidates generated by the 14 patterns of item-based CF and 2 patterns of user-based CF were assembled to create the final candidates.

Since **sessions with a larger number of aids in the test data are more likely to have a larger number of events in the forecasting period**, more candidates were created for sessions with a larger number of aids in the test data.

def get_aid_similarity1(df, topk=200):
    session_info = df.drop_duplicates(["session", "aid"], keep="last")\
                     .groupby("session", as_index=False)[["aid", "type", "ts"]].agg(list) 
    
    aid_similarity = {}
    for session, aids, tps, tss in tqdm(zip(session_info["session"],
                                            session_info["aid"],
                                            session_info["type"],
                                            session_info["ts"]),
                                        total=len(session_info)):
        for aid1, tp1, ts1 in zip(aids, tps, tss):
            session_length = math.sqrt(len(aids))
            aid_similarity.setdefault(aid1, Counter())
            for aid2, tp2, ts2 in zip(aids, tps, tss):
                if (aid1 == aid2):
                    continue
                aid_similarity[aid1][aid2] += (1/session_length)
    
    # Exclude all but the topK to save time and memory
    for aid1, aid2_dict in tqdm(aid_similarity.items()):
        relations = dict(aid2_dict.most_common(topk))
        # normalize
        if len(relations) == 0:
            continue
        max_num = relations[max(relations, key=relations.get)]
        if max_num == 0:
            continue
        aid_similarity[aid1] = {k: v / max_num for k, v in relations.items()}

    del session_info; gc_clear()
    return aid_similarity
```

def make_real_session(df, hours=2):
    df["lag"] = df["ts"] - df.groupby("session")["ts"].shift(1)
    df["real_session"] = (df["lag"] > 1000*60*60*hours).astype('int8').fillna(0)
    df["real_session"] = df.groupby("session")["real_session"].cumsum()
    del df["lag"]; gc_clear()
    return df

def get_aid_similarity2(df, topk=200):

    df = make_real_session(df, hours=4)
    session_info = df.groupby(["session", "real_session"], as_index=False)[["aid", "type", "ts"]].agg(list) 
    
    aid_similarity = {}
    aid_cnt = defaultdict(int)
    for session, real_session, aids, tps, tss in tqdm(zip(session_info["session"],
                                                          session_info["real_session"],
                                                          session_info["aid"],
                                                          session_info["type"],
                                                          session_info["ts"]),
                                                      total=len(session_info)):
        for aid1, tp1, ts1 in zip(aids, tps, tss):
            aid_similarity.setdefault(aid1, Counter())
            for aid2, tp2, ts2 in zip(aids, tps, tss):
                if (abs(ts1-ts2)>24*60*60*1000) or (aid1 == aid2):
                    continue
                aid_cnt[aid1] += 1
                if min(tp1, tp2)==0:
                    aid_similarity[aid1][aid2] += 1
                elif min(tp1, tp2)==1:
                    aid_similarity[aid1][aid2] += 3
                elif min(tp1, tp2)==2:
                    aid_similarity[aid1][aid2] += 6
    
    # Exclude all but the topK to save time and memory
    for aid1, aid2_dict in tqdm(aid_similarity.items()):  
        for aid2, score in aid2_dict.items():  
            aid_similarity[aid1][aid2] = score / math.sqrt(aid_cnt[aid1]*aid_cnt[aid2])
        aid_similarity[aid1] = dict(aid2_dict.most_common(topk))
       
    del session_info; gc_clear()
    return aid_similarity
```

```
def recommend_aid(df,
                  _test_df,
                  target_session,
                  aid_similarity,
                  feature_name,
                  type_weighted=False, type_weights={0:1, 1:3, 2:6},
                  time_weighted=False,
                  only_last_aid=False,
                  same_real_session=False, real_session_hours=2,
                  drop_duplicates=False
                  ):
    
    # ==============================================================
    #  test_df preprocess
    # ==============================================================

    if only_last_aid:
        _test_df = _test_df.groupby("session").last().reset_index()

    if same_real_session:
        _test_df = make_real_session(_test_df, hours=real_session_hours)
        _test_df["real_session_max"] = _test_df.groupby("session")["real_session"].transform("max")
        _test_df = _test_df[_test_df["real_session"]==_test_df["real_session_max"]].reset_index(drop=True)
        del _test_df["lag"], _test_df["real_session"], _test_df["real_session_max"]; gc_clear()
    
    if drop_duplicates:
        _test_df = _test_df.groupby(["session", "aid"], as_index=False)[["type", "ts"]].max()
    
    # ==============================================================
    #  Create recommend dictionary
    # ==============================================================
        
    if type_weighted:
        
        _test_df["type_weights"] = _test_df["type"].map(type_weights).astype("int8")
        session_info = _test_df[_test_df["session"].isin(target_session)]\
                       .groupby("session", as_index=False)[["aid", "type_weights", "ts"]].agg(list) 
        del _test_df; gc_clear()

        rec = {}
        for session, aids, type_weights, tss in tqdm(zip(session_info["session"],
                                                         session_info["aid"],
                                                         session_info["type_weights"],
                                                         session_info["ts"]),
                                                     total=len(session_info)):
            rec.setdefault(session, Counter())
            if time_weighted:
                time_weights = make_time_weights(tss)
                for aid, time_weight, type_weight in zip(aids, time_weights, type_weights):
                    rec[session] += {aid: v*time_weight*type_weight for (aid, v) in aid_similarity.get(aid, {}).items()}
            else:
                for aid, type_weight in zip(aids, type_weights):
                    rec[session] += {aid: v*type_weight for (aid, v) in aid_similarity.get(aid, {}).items()}

    else:
        session_info = _test_df[_test_df["session"].isin(target_session)]\
                       .groupby("session", as_index=False)[["aid", "ts"]].agg(list) 
        del _test_df; gc_clear()

        rec = {}
        for session, aids, tss in tqdm(zip(session_info["session"],
                                           session_info["aid"],
                                           session_info["ts"]),
                                       total=len(session_info)):
            rec.setdefault(session, Counter())
            if time_weighted:
                time_weights = make_time_weights(tss)
                for aid, time_weight in zip(aids, time_weights):
                    rec[session] += {aid: v*time_weight for (aid, v) in aid_similarity.get(aid, {}).items()}
            else:
                for aid in aids:
                    rec[session] += aid_similarity.get(aid, {})        
    del session_info; gc_clear()
    
    # ==============================================================
    #  Create features
    # ==============================================================
    feature = []
    for session, aid in tqdm(zip(df["session"], df["aid"]), total=len(df)):
        feature.append( rec.get(session, {}).get(aid, 0) )
    df[feature_name] = feature

    del feature, rec; gc_clear()
    return df
```

# Model training

Different pipelines were used depending on which of the orders/carts/clicks was the target.

For orders and carts, models were trained in two stages (1st stage/2nd stage).


#### orders model
- **1st stage**

All candidates were trained on the model without negative sampling.
I didn't want to do negative sampling as much as possible because negative sampling lowers the score.

- **2nd stage**

As a result of the 1st stage, only the top 50 candidates per session were selected for the 2nd stage.
The predictions from the 1st stage were not used for the 2nd stage features.

#### carts model
- **1st stage**

Since I could not run the model on all the data without negative sampling, I trained the model with negative sampling (x 0.3) in the 1st stage.

- **2nd stage**

The model of the 1st stage was used to create oof for all candidates, and then the top 50 candidates per session were selected for the 2nd stage.

#### clicks model
- **1st stage**

Only the 1st stage was run with negative sampling (x 0.1).

Probably the score would be higher if the 2nd stage was performed as in the carts model, but since the clicks have less weight on the score, the 2nd stage for the clicks was not conducted for cost-effectiveness.

#### Algorithm and Parameters

I created one Catboost model each to predict clicks, carts, and orders.

# Ensemble

I created models with 5 seeds and ensembled them.
(I changed not only the seed of the model parameters, but also the seed in the test/ground truth split.)

Unlike the ensemble method in [@radek1's notebook](https://www.kaggle.com/code/radek1/2-methods-how-to-ensemble-predictions), the ensemble was performed in a way to apply weight to each of the predicted rankings.
